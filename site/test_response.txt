  <script type="module">
  import * as THREE from 'three'

  const container = document.getElementById('canvas3d')
  const scene = new THREE.Scene()
  const camera = new THREE.PerspectiveCamera(55, window.innerWidth / window.innerHeight, 0.1, 120)
  camera.position.z = 32

  const renderer = new THREE.WebGLRenderer({ alpha: true, antialias: true })
  renderer.setSize(window.innerWidth, window.innerHeight)
  renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2))
  renderer.setClearColor(0x000000, 0)
  container.appendChild(renderer.domElement)

  const W = 22, H = 14, D = 20

  const texts = [
    "{ C }", "{ C++ }", "{ C# }", "{ Python }", "{ Rust }",
    "{ x86 }", "{ x64 }", "{ ARM }",
    "root", "syscall", "0x7F", "segfault", "shellcode",
    "PWN", "DEBUG", "REVERSE", "HACK", "crack",
    "buffer overflow", "ROP chain", "null byte",
    "exploit", "disasm", "patch", "inject", "dump",
    "trace", "breakpoint", "analyse", "kernel",
  ]

  function makeTextSprite(text) {
    const canvas = document.createElement('canvas')
    const ctx = canvas.getContext('2d')
    const isLang = text.startsWith('{')
    const fontSize = isLang ? 48 : 34
    ctx.font = `bold ${fontSize}px "JetBrains Mono", monospace`
    const metrics = ctx.measureText(text)
    const w = metrics.width + 40
    const h = fontSize + 40
    canvas.width = w * 2
    canvas.height = h * 2
    ctx.scale(2, 2)

    ctx.clearRect(0, 0, w, h)
    ctx.font = `bold ${fontSize}px "JetBrains Mono", monospace`
    ctx.textAlign = 'center'
    ctx.textBaseline = 'middle'

    ctx.shadowColor = 'rgba(96,165,250,0.5)'
    ctx.shadowBlur = 14
    ctx.fillStyle = isLang ? 'rgba(96,165,250,0.8)' : 'rgba(255,255,255,0.4)'
    ctx.fillText(text, w / 2, h / 2)
    ctx.shadowBlur = 24
    ctx.fillText(text, w / 2, h / 2)

    const tex = new THREE.CanvasTexture(canvas)
    tex.needsUpdate = true
    const mat = new THREE.SpriteMaterial({
      map: tex,
      transparent: true,
      depthWrite: false,
      blending: THREE.AdditiveBlending,
      opacity: 0.8,
    })
    const sprite = new THREE.Sprite(mat)
    const aspect = w / h
    const baseScale = isLang ? 2.2 : 1.5
    sprite.scale.set(baseScale * aspect, baseScale, 1)
    return sprite
  }

  const sprites = []
  for (let i = 0; i < texts.length + 10; i++) {
    const text = texts[i % texts.length]
    const sprite = makeTextSprite(text)
    sprite.position.set(
      (Math.random() - 0.5) * W * 2,
      (Math.random() - 0.5) * H * 2,
      (Math.random() - 0.5) * D * 2
    )
    const speed = 0.06 + Math.random() * 0.14
    const angle = Math.random() * Math.PI * 2
    const angle2 = Math.random() * Math.PI * 2
    sprite.userData = {
      vx: Math.cos(angle) * Math.cos(angle2) * speed,
      vy: Math.sin(angle2) * speed,
      vz: Math.sin(angle) * Math.cos(angle2) * speed,
      phase: Math.random() * 100,
      wanderSpeed: 0.08 + Math.random() * 0.16,
    }
    scene.add(sprite)
    sprites.push(sprite)
  }

  let scrollY = 0
  window.addEventListener('scroll', () => {
    scrollY = window.scrollY / (document.documentElement.scrollHeight - window.innerHeight)
  }, { passive: true })

  window.addEventListener('resize', () => {
    camera.aspect = window.innerWidth / window.innerHeight
    camera.updateProjectionMatrix()
    renderer.setSize(window.innerWidth, window.innerHeight)
  })

  function animate() {
    requestAnimationFrame(animate)
    const t = Date.now() * 0.001
    const energy = 1 + scrollY * 0.4
    const dt = 0.016

    sprites.forEach((sprite, i) => {
      const ud = sprite.userData
      const w1 = Math.sin(t * ud.wanderSpeed + ud.phase + i * 0.5)
      const w2 = Math.cos(t * ud.wanderSpeed * 0.6 + ud.phase + i * 0.7)
      const w3 = Math.sin(t * ud.wanderSpeed * 0.4 + ud.phase + i * 0.3)

      sprite.position.x += (ud.vx + w1 * 0.12) * dt * energy
      sprite.position.y += (ud.vy + w2 * 0.12) * dt * energy
      sprite.position.z += (ud.vz + w3 * 0.12) * dt * energy

      if (sprite.position.x > W) { sprite.position.x = W; ud.vx *= -1 }
      if (sprite.position.x < -W) { sprite.position.x = -W; ud.vx *= -1 }
      if (sprite.position.y > H) { sprite.position.y = H; ud.vy *= -1 }
      if (sprite.position.y < -H) { sprite.position.y = -H; ud.vy *= -1 }
      if (sprite.position.z > D) { sprite.position.z = D; ud.vz *= -1 }
      if (sprite.position.z < -D) { sprite.position.z = -D; ud.vz *= -1 }
    })

    renderer.render(scene, camera)
  }
  animate()
  </script>

  function copyCmd(btn, text) {
    navigator.clipboard.writeText(text).then(() => {
      btn.textContent = 'Copiado!';
      btn.classList.add('copied');
      setTimeout(() => { btn.textContent = 'Copiar'; btn.classList.remove('copied'); }, 2000);
    });
  }
